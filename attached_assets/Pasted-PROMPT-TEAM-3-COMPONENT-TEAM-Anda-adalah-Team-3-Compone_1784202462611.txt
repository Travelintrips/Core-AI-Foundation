menentukan jenis komponen;
menentukan region;

Output:

interface ComponentPlan {
  components: Array<{
    id: string;
    sectionId: string;
    type: string;
    role: string;
    required: boolean;
    contentSource:
      | "static"
      | "variable"
      | "asset"
      | "generated-placeholder";
    bindingKey?: string;
    region: {
      x: number;
      y: number;
      width: number;
      height: number;

      | "foreground";

interface VariablePlan {
  variables: Array<{
    key: string;
    label: string;
    type:
      | "text"
      | "multiline"
      | "number"
      | "currency"
      | "url"
      | "phone"
      | "email"
      | "date";
    required: boolean;
    defaultValue?: string | number;
    placeholder?: string;
    validation?: {
      minLength?: number;
      maxLength?: number;
      min?: number;

interface AssetPlan {

    id: string;
    type:
      | "photo"
      | "logo"
      | "icon"
      | "background"
      | "illustration"
      | "qr";
    componentId: string;
    purpose: string;

      width: number;
      height: number;
    };

    cropFocus?: "center" | "top" | "bottom" | "left" | "right";
    acceptedMimeTypes?: string[];

  }>;
}
